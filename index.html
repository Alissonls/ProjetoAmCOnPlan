Integração Amazon Connect

Portas de comunicação ativas para Contact Flows, ChatJS e WebSockets AWS

Modo Amazon Connect Live Roteia mensagens e eventos diretamente para as filas do seu Contact Center AWS
URL da sua função AWS Lambda que executa a API StartChatContact do Amazon Connect.
📋 Ver modelo da função AWS Lambda (Node.js) para StartChatContact 
// AWS Lambda (Node.js 18+) para iniciar chat no Amazon Connect
const { ConnectClient, StartChatContactCommand } = require("@aws-sdk/client-connect");
const connect = new ConnectClient({ region: process.env.AWS_REGION || "us-east-1" });

exports.handler = async (event) => {
  const body = JSON.parse(event.body || "{}");
  const params = {
    InstanceId: process.env.CONNECT_INSTANCE_ID || body.InstanceId,
    ContactFlowId: process.env.CONNECT_CONTACT_FLOW_ID || body.ContactFlowId,
    ParticipantDetails: { DisplayName: body.ParticipantDetails?.DisplayName || "Cliente" },
    Attributes: body.Attributes || {},
    InitialMessage: body.InitialMessage
  };
  const command = new StartChatContactCommand(params);
  const response = await connect.send(command);

  return {
    statusCode: 200,
    headers: {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Headers": "Content-Type",
      "Access-Control-Allow-Methods": "POST,OPTIONS"
    },
    body: JSON.stringify({
      ContactId: response.ContactId,
      ParticipantId: response.ParticipantId,
      ParticipantToken: response.ParticipantToken
    })
  };
};
